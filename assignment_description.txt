tcp_nice.c - TCP Nice Implementation (this is an example)
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/tcp.h>

// you have to modify for your own tcp nice congestion control algorithm
 static void tcp_nice_cong_avoid(struct sock *sk, u32 ack, u32 acked) {
    struct tcp_sock *tp = tcp_sk(sk);
    if (!tcp_is_cwnd_limited(sk))
        return;
    
    printk(KERN_INFO "TCP Nice: cwnd=%u, ssthresh=%u\n", tp->snd_cwnd, tp->snd_ssthresh);

    if (tp->snd_cwnd < tp->snd_ssthresh)
        tp->snd_cwnd++;  // Slow start
    else
        tp->snd_cwnd = tcp_cong_avoid_ai(tp, tcp_snd_cwnd(tp), acked); // Reno-like behavior 
}

static struct tcp_congestion_ops tcp_nice = {
    .flags      = TCP_CONG_NON_RESTRICTED,
    .name       = "nice",
    .cong_avoid = tcp_nice_cong_avoid,
};

static int __init tcp_nice_register(void) {
    return tcp_register_congestion_control(&tcp_nice);
}

static void __exit tcp_nice_unregister(void) {
    tcp_unregister_congestion_control(&tcp_nice);
}

module_init(tcp_nice_register);
module_exit(tcp_nice_unregister);

Using TCP Nice in Applications
iperf3 Example
To test TCP Nice with iperf3, use the following command:

iperf3 -c <server_ip> --congestion nice
To set TCP Nice in C code using setsockopt():

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <netinet/tcp.h>

int main() {
    int sockfd;
    char cc_algo[] = "nice";

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd == -1) {
        perror("socket");
        exit(EXIT_FAILURE);
    }

    if (setsockopt(sockfd, IPPROTO_TCP, TCP_CONGESTION, cc_algo, strlen(cc_algo)) < 0) {
        perror("setsockopt");
        close(sockfd);
        exit(EXIT_FAILURE);
    }

    printf("Successfully set congestion control to: %s\n", cc_algo);

    char new_cc_algo[16];
    socklen_t len = sizeof(new_cc_algo);
    if (getsockopt(sockfd, IPPROTO_TCP, TCP_CONGESTION, new_cc_algo, &len) < 0) {
        perror("getsockopt");
        close(sockfd);
        exit(EXIT_FAILURE);
    }

    printf("Current congestion control algorithm: %s\n", new_cc_algo);
    close(sockfd);
    return 0;
}
Tracking the cwnd and ssthresh variables in the Linux Kernel
There are multiple ways to track the cwnd and ssthresh variables of a given TCP flow:

Using the tcp_info Structure:
You can retrieve cwnd and ssthresh values directly in your program using the tcp_info structure. The following code provides an example of how to obtain these values for a given socket. You can call this function while your program is sending data to the other endpoint.
The easiest approach is to write a TCP client that continuously sends data to a server after setting the TCP congestion control algorithm using setsockopt(). You can then periodically retrieve cwnd and ssthresh values. Below is an example of how to use tcp_info to obtain these parameters for a TCP socket.


#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <netinet/tcp.h>
#include <sys/socket.h>

#define SERVER_IP "127.0.0.1"  // Change to the target server's IP
#define SERVER_PORT 8080       // Change to the target server's port

void print_tcp_info(int sockfd) {
    struct tcp_info info;
    socklen_t len = sizeof(info);
    
    if (getsockopt(sockfd, IPPROTO_TCP, TCP_INFO, &info, &len) == 0) {
        printf("CWND: %u, SSTHRESH: %u\n", info.tcpi_snd_cwnd, info.tcpi_snd_ssthresh);
    } else {
        perror("getsockopt");
    }
}

int main() {
    int sockfd;
    struct sockaddr_in server_addr;

    // Create socket
    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) {
        perror("Socket creation failed");
        exit(EXIT_FAILURE);
    }

    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(SERVER_PORT);
    if (inet_pton(AF_INET, SERVER_IP, &server_addr.sin_addr) <= 0) {
        perror("Invalid address");
        close(sockfd);
        exit(EXIT_FAILURE);
    }

    // Connect to server
    if (connect(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0) {
        perror("Connection failed");
        close(sockfd);
        exit(EXIT_FAILURE);
    }

    printf("Connected to server. Tracking CWND and SSTHRESH...\n");

    // While sending the data, you can track cwnd and ssthresh periodically.
    // read/write code is not present.
    for (int i = 0; i < 10; i++) {
        print_tcp_info(sockfd);
        sleep(1);
    }

    close(sockfd);
    return 0;
}

Using perf with tcp_probe:
The perf tool, combined with tcp_probe, allows tracking of kernel-level TCP variables in real time. While this method may be slightly outdated, it remains a useful way to monitor congestion control behavior without modifying the Linux kernel’s congestion control algorithm.

 

Kernel-Level Logging (Least Preferred):
A less preferred method is modifying the Linux kernel module to print tp->snd_cwnd and tp->snd_ssthresh using printk(). This requires modifying and recompiling the kernel, which is more complex. You can refer to the tcp_nice source code example above for reference.

Network Emulation Using NetEm
The following is an example configuration, and you should modify the settings as necessary:

sudo tc qdisc add dev eth0 root handle 1: htb default 12
sudo tc class add dev eth0 parent 1: classid 1:12 htb rate 100mbit
sudo tc qdisc add dev eth0 parent 1:12 handle 20: netem delay 100ms
sudo tc qdisc add dev eth0 parent 20: handle 30: bfifo limit 1000k
Explanation
tc qdisc add dev eth0 root handle 1: htb default 12: Sets HTB (Hierarchical Token Bucket) for bandwidth control.
tc class add dev eth0 parent 1: classid 1:12 htb rate 100mbit: Limits bandwidth to 100 Mbps.
tc qdisc add dev eth0 parent 1:12 handle 20: netem delay 100ms: Adds 100 ms delay.
tc qdisc add dev eth0 parent 20: handle 30: bfifo limit 1000k: Sets a 1000 KB buffer size.
To verify:

sudo tc -s qdisc show dev eth0
To remove settings:

sudo tc qdisc del dev eth0 root
